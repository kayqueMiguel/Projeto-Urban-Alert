-- =====================================================
-- USUARIOS
-- =====================================================

CREATE TABLE usuarios (

    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,

    nome VARCHAR(100) NOT NULL,

    email VARCHAR(150) UNIQUE NOT NULL,

    senha VARCHAR(255) NOT NULL,

    telefone VARCHAR(20),

    tipo_usuario VARCHAR(20)
    DEFAULT 'CIDADAO'
    CHECK (
        tipo_usuario IN (
            'CIDADAO',
            'ADMIN',
            'PREFEITURA'
        )
    ),

    ativo BOOLEAN DEFAULT TRUE,

    data_criacao TIMESTAMPTZ DEFAULT NOW()

);

-- =====================================================
-- TIPOS DE PROBLEMA
-- =====================================================

CREATE TABLE tipos_problema (

    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,

    nome VARCHAR(100) NOT NULL,

    descricao TEXT

);

-- =====================================================
-- STATUS
-- =====================================================

CREATE TABLE status_denuncia (

    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,

    nome VARCHAR(50) NOT NULL,

    descricao TEXT

);

-- =====================================================
-- DENUNCIAS
-- =====================================================

CREATE TABLE denuncias (

    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,

    usuario_id BIGINT,

    tipo_problema_id BIGINT NOT NULL,

    status_id BIGINT NOT NULL DEFAULT 1,

    titulo VARCHAR(150) NOT NULL,

    endereco VARCHAR(255) NOT NULL,

    bairro VARCHAR(100),

    cidade VARCHAR(100),

    descricao TEXT NOT NULL,

    latitude NUMERIC(10,7),

    longitude NUMERIC(10,7),

    anonima BOOLEAN DEFAULT FALSE,

    prioridade VARCHAR(10)
    DEFAULT 'MEDIA'
    CHECK (
        prioridade IN (
            'BAIXA',
            'MEDIA',
            'ALTA'
        )
    ),

    data_criacao TIMESTAMPTZ DEFAULT NOW(),

    data_atualizacao TIMESTAMPTZ DEFAULT NOW(),

    CONSTRAINT fk_denuncia_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id),

    CONSTRAINT fk_denuncia_tipo
        FOREIGN KEY (tipo_problema_id)
        REFERENCES tipos_problema(id),

    CONSTRAINT fk_denuncia_status
        FOREIGN KEY (status_id)
        REFERENCES status_denuncia(id)

);

-- =====================================================
-- IMAGENS
-- =====================================================

CREATE TABLE imagens_denuncia (

    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,

    denuncia_id BIGINT NOT NULL,

    nome_arquivo VARCHAR(255),

    url_imagem TEXT NOT NULL,

    data_upload TIMESTAMPTZ DEFAULT NOW(),

    CONSTRAINT fk_imagem_denuncia
        FOREIGN KEY (denuncia_id)
        REFERENCES denuncias(id)
        ON DELETE CASCADE

);

-- =====================================================
-- COMENTARIOS
-- =====================================================

CREATE TABLE comentarios_denuncia (

    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,

    denuncia_id BIGINT NOT NULL,

    usuario_id BIGINT NOT NULL,

    comentario TEXT NOT NULL,

    data_comentario TIMESTAMPTZ DEFAULT NOW(),

    CONSTRAINT fk_comentario_denuncia
        FOREIGN KEY (denuncia_id)
        REFERENCES denuncias(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_comentario_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)

);

-- =====================================================
-- HISTORICO
-- =====================================================

CREATE TABLE historico_status (

    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,

    denuncia_id BIGINT NOT NULL,

    status_id BIGINT NOT NULL,

    usuario_id BIGINT,

    observacao TEXT,

    data_alteracao TIMESTAMPTZ DEFAULT NOW(),

    CONSTRAINT fk_historico_denuncia
        FOREIGN KEY (denuncia_id)
        REFERENCES denuncias(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_historico_status
        FOREIGN KEY (status_id)
        REFERENCES status_denuncia(id),

    CONSTRAINT fk_historico_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)

);

-- =====================================================
-- CONTATOS
-- =====================================================

CREATE TABLE contatos (

    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,

    nome VARCHAR(100),

    email VARCHAR(150),

    assunto VARCHAR(150),

    mensagem TEXT,

    status_contato VARCHAR(20)
    DEFAULT 'ABERTO'
    CHECK (
        status_contato IN (
            'ABERTO',
            'RESPONDIDO'
        )
    ),

    data_envio TIMESTAMPTZ DEFAULT NOW()

);
